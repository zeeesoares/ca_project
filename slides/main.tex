\section{Introduction}

\subsection{Context}
\begin{frame}{Context}
    \justifying
    Training Large Language Models (LLMs) involves refining billions of parameters over several weeks of continuous computation.
    
    \vspace{1em}
    To ensure resilience against hardware failures and prevent total progress loss, model states must be periodically saved to a Parallel File System (PFS).
    
    \vspace{1em}
    The challenge lies in optimizing this persistence process without compromising data integrity or resources utilization.
\end{frame}

\subsection{Problem}
\begin{frame}{Problem: The Checkpointing Overhead}
    \centering
    \small
    \begin{tabular}{l p{0.6\textwidth}}
        \hline
        \textbf{Challenge} & \textbf{Impact} \\ \hline
        \noalign{\smallskip}
        PFS Contention & Simultaneous writes from multiple nodes saturate the shared storage bandwidth. \\ 
        \noalign{\smallskip}
        Training Stalls & Heavy I/O operations block the training process, leading to significant idling. \\ 
        \noalign{\smallskip}
        Noisy Neighbor & Large-scale checkpointing bursts degrade performance for all concurrent HPC jobs.  \\
        \noalign{\smallskip}
        Storage Costs & Frequent writes escalate operational overhead and costs as model size grows. \\
        \hline
    \end{tabular}
    
    \vspace{1em}
    \footnotesize
    \textit{Table 1: Technical bottlenecks of direct checkpointing to PFS.}
\end{frame}

\subsection{Motivation}
\begin{frame}{Motivation}
    \small
    The goal is to design a checkpointing system that:
    
    \centering
    \begin{itemize}
        \small
        \item Mitigates I/O contention on the PFS, ensuring fair resource sharing and Cluster Harmony.
        \item Provides resilience against hardware failures, minimizing training disruptions.
        \item Optimizes checkpointing process and storage strategies to balance performance and costs.
    \end{itemize}
\end{frame}

\section{Approach and Architecture}
\subsection{Approach}
\begin{frame}{Approach: Policy-Driven Checkpointing}
    \centering
    \small

    \begin{columns}[t]
        \begin{column}{0.31\textwidth}
            \begin{block}{Tiered Persistence}
                \footnotesize \textbf{Local Storage} as a buffer, decoupling training from PFS latency.
            \end{block}
        \end{column}
        \begin{column}{0.31\textwidth}
            \begin{block}{Controlled PFS Access}
                \footnotesize Mitigate I/O contention via a centralized \textbf{Orchestrator}.
            \end{block}
        \end{column}
    \end{columns}
    
    \vspace{1em}
    
    \begin{columns}[t]
        \begin{column}{0.31\textwidth}
            \begin{block}{Policy-Driven}
                \footnotesize Dynamic migration based on cluster-wide scheduling policies.
            \end{block}
        \end{column}
        \begin{column}{0.31\textwidth}
            \begin{block}{Traffic Shaping}
                \footnotesize \textbf{Token Bucket} to smooth traffic and eliminate noisy neighbors.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
